import { marchProgress, type ResolvedMarch } from '../timeline/model'

/** The place a unit fires on between two minutes (in world coordinates, with its place id), for turning toward it. */
export type Aim = { from: number; to: number; x: number; z: number; target?: string; /** The unit whose attack this is (the guns keep their shells apart by unit). */ unit?: string }
/** A direction on the ground (not necessarily a unit vector). */
export type Heading = { x: number; z: number }

const TAU = Math.PI * 2

/** An angle brought into (-π, π]. */
export function wrapPi(a: number): number {
  const r = a - TAU * Math.round(a / TAU)
  return r <= -Math.PI ? r + TAU : r
}

/**
 * The angle `atan2(x, z)` of a ground direction. Turning an object by `yaw` about the vertical axis (three.js `rotation.y`) adds
 * `yaw` to the angle of everything on it, so a model whose front has angle `front` faces `heading` when turned by `yawToFace`.
 */
export const bearing = (x: number, z: number) => Math.atan2(x, z)
export const yawToFace = (heading: number, front: number) => wrapPi(heading - front)

/** Moves angle `current` toward `target` the short way round, closing the gap at about `rate` per second (a smooth turn, whatever the frame time). */
export function dampAngle(current: number, target: number, rate: number, dt: number): number {
  return current + wrapPi(target - current) * (1 - Math.exp(-rate * dt))
}

/** The aim that is on at minute `t`; when several are, the one that began last. */
export function aimAt(aims: readonly Aim[], t: number): Aim | null {
  let best: Aim | null = null
  for (const a of aims) if (t >= a.from && t <= a.to && (!best || a.from >= best.from)) best = a
  return best
}

/**
 * The aim of one gun (`slot`) of a battery at minute `t`. While several aims are on, the guns share them out in turn (slot 0 takes the
 * first, slot 1 the second, and so on round), so a battery fires on every target at once; with none on, a gun keeps facing the last place
 * it fired on, or the first it will fire on.
 */
export function gunAim(aims: readonly Aim[], t: number, slot: number): Aim | null {
  const on = aims.filter((a) => t >= a.from && t <= a.to)
  if (on.length > 0) return on[slot % on.length]
  let last: Aim | null = null
  for (const a of aims) if (a.to < t && (!last || a.to >= last.to)) last = a
  if (last) return last
  let next: Aim | null = null
  for (const a of aims) if (a.from > t && (!next || a.from < next.from)) next = a
  return next
}

/** The way a path runs (on the ground) `fraction` of the way along it, by length; null for a path with no length. */
export function pathHeading(path: readonly { x: number; z: number }[], fraction: number): Heading | null {
  let total = 0
  for (let i = 1; i < path.length; i++) total += Math.hypot(path[i].x - path[i - 1].x, path[i].z - path[i - 1].z)
  if (total <= 0) return null
  const want = Math.min(Math.max(fraction, 0), 1) * total
  let done = 0
  let last: Heading | null = null
  for (let i = 1; i < path.length; i++) {
    const dx = path[i].x - path[i - 1].x, dz = path[i].z - path[i - 1].z
    const len = Math.hypot(dx, dz)
    if (len <= 0) continue
    last = { x: dx / len, z: dz / len }
    if (done + len >= want) return last
    done += len
  }
  return last
}

/** Nearer than this (world units) to the place it fires on, a unit has no direction to turn to. */
const NO_DIRECTION = 1

/**
 * The way a unit faces at minute `t`, standing at `at`, given its own marches (any order) and aims:
 * - along its path while it marches;
 * - toward the place it fires on while it is firing;
 * - otherwise the way it last walked;
 * - before it has done anything, toward whatever it does first (so a group waiting to fire already faces its target).
 * Null when there is nothing to go by: the statue keeps the way it was baked.
 */
export function headingAt(marches: readonly ResolvedMarch[], aims: readonly Aim[], t: number, at: { x: number; z: number }): Heading | null {
  const walking = marches.find((m) => t >= m.start && t < m.end)
  if (walking) return pathHeading(walking.path, marchProgress(walking, t))
  const aim = aimAt(aims, t)
  if (aim) {
    const dx = aim.x - at.x, dz = aim.z - at.z
    if (Math.hypot(dx, dz) >= NO_DIRECTION) return { x: dx, z: dz }
  }
  let finished: ResolvedMarch | null = null
  for (const m of marches) if (t >= m.end && (!finished || m.end >= finished.end)) finished = m
  if (finished) return pathHeading(finished.path, 1)
  // Not yet active: the first march or aim still to come.
  let nextMarch: ResolvedMarch | null = null
  for (const m of marches) if (t < m.start && (!nextMarch || m.start < nextMarch.start)) nextMarch = m
  let nextAim: Aim | null = null
  for (const a of aims) if (t < a.from && (!nextAim || a.from < nextAim.from)) nextAim = a
  if (nextAim && (!nextMarch || nextAim.from <= nextMarch.start)) {
    const dx = nextAim.x - at.x, dz = nextAim.z - at.z
    return Math.hypot(dx, dz) >= NO_DIRECTION ? { x: dx, z: dz } : null
  }
  return nextMarch ? pathHeading(nextMarch.path, 0) : null
}

/** How a group of soldiers rocks on the march: the hop of its steps (world units), a sway and a nod (radians). All zero when `walk` is 0 (standing). */
export type Rock = { hop: number; roll: number; pitch: number }

/** Steps per second of a marching group (radians per second of the hop's phase), and how much it hops and sways. */
const STEP_RATE = 8
const HOP = 1.3
const SWAY = 0.045
const NOD = 0.03

/** The rock of a group `walk` (0..1) of the way into a march at `time` seconds; `seed` keeps groups out of step. */
export function marchRock(time: number, seed: number, walk: number): Rock {
  const phase = time * STEP_RATE + seed * 2.3
  return {
    hop: Math.abs(Math.sin(phase)) * HOP * walk,
    roll: Math.sin(phase * 0.5) * SWAY * walk,
    pitch: Math.sin(phase * 0.5 + 1.1) * NOD * walk,
  }
}
